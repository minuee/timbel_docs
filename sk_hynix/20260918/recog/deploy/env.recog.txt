# recog-backend — runtime environment (copy to .env.recog on the deploy host)
#
# CI updates RECOG_VERSION in place via `sed`.
# All other values are managed by hand on the deploy host.

# ---------------------------------------------------------------------
# Image / version — auto-updated by CI deploy stage
# ---------------------------------------------------------------------
# 오프라인 설치 시 install_offline.sh 가 이 두 값을 로드한 이미지로 덮어쓴다.
IMAGE_NAME=ghcr.io/timbel-timblo-onpremise/recog-backend
RECOG_VERSION=skHynix-latest

# missing = 없을 때만 pull (CI) / never = 절대 pull 안 함 (망분리)
PULL_POLICY=never

# ---------------------------------------------------------------------
# Ports
#   HOST_PORT       — port exposed on the deploy machine
#   CONTAINER_PORT  — port the service binds inside the container
# ---------------------------------------------------------------------
HOST_PORT=28080
CONTAINER_PORT=8080

# ---------------------------------------------------------------------
# Runtime bind (inside the container — read by recog via env)
# ---------------------------------------------------------------------
RECOG_HOST=0.0.0.0
RECOG_PORT=8080

# ---------------------------------------------------------------------
# 데이터 저장 위치 (호스트 경로)
#   세션 데이터 + 병합 작업 기록(merge.db)이 여기에 쌓인다.
#   이미지를 다시 빌드해도 이 디렉터리는 건드리지 않는다.
#
#   기본값 ./runtime 은 compose 파일 기준 상대경로다.
#   소스를 통째로 덮어쓰는 방식으로 배포한다면 소스 밖 절대경로를 권장한다.
#     RUNTIME_DIR=/var/lib/recog-runtime
# ---------------------------------------------------------------------
RUNTIME_DIR=/data/timblo/recog-runtime

# ---------------------------------------------------------------------
# API 문서(/docs, /openapi.json) 접근 계정
#   브라우저에서 여는 유일한 화면이라 Basic 인증을 건다.
#   /v1/merge 엔드포인트 자체는 인증이 없다 (도커 네트워크 내부 통신).
#   SWAGGER_PASSWORD 를 빈 값으로 두면 인증이 꺼진다.
# ---------------------------------------------------------------------
SWAGGER_USER=admin
SWAGGER_PASSWORD=lena47

# ---------------------------------------------------------------------
# 음성 병합 API (/v1/merge)
#   전부 선택값이다. 비워두면 아래 기본값으로 동작한다.
# ---------------------------------------------------------------------
# 동시에 실제로 병합을 돌리는 개수. 요청 접수는 이 값과 무관하게 무제한이다.
MERGE_WORKERS=3
# 한 건당 최대 파일 수
MERGE_MAX_FILES=5
# presigned 경로의 파일당 상한 (bytes, 기본 2GB)
MERGE_MAX_FILE_BYTES=2147483648
# 업로드 테스트 경로(/v1/merge/upload)의 합계 상한 (bytes, 기본 512MB)
#   이 경로는 본문을 메모리에 버퍼링하므로 낮게 잡는다.
MERGE_UPLOAD_MAX_BYTES=536870912
# RUNNING 진입 후 제한시간 (초). 큐 대기 시간은 포함되지 않는다.
MERGE_JOB_TIMEOUT_SEC=1800
# align=timestamp에서 이 값을 넘는 offset은 시계 오류로 보고 0으로 되돌린다 (ms)
MERGE_MAX_OFFSET_MS=60000
# 작업 상태를 메모리에 캐시하는 기간 (초, 기본 24시간)
#   이 값이 지나도 조회는 된다 — SQLite 기록으로 자동 폴백한다.
MERGE_JOB_TTL_SEC=86400
# SQLite 기록 보존 기간 (일, 기본 90일)
#   runtime/merge.db 에 저장되며 재시작해도 유지된다.
#   0 으로 두면 영구 보관.
MERGE_RECORD_TTL_DAYS=90
