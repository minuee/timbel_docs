#!/usr/bin/env bash
#
# howling_test_007.sh — 0.0.7(파형 맞추기) 하울링 재시험: 회의마다 결과 파일 1개.
#
# 시험 조건 (하나):
#   파형 맞추기 켬(ALIGN_REFINE=1), 음량 맞추기 끔(MERGE_LEVEL=off), 오토믹서 켬(지금 설정 그대로)
#
# 사용법 (회사 서버에서, 0.0.7 설치 후):
#   ./howling_test_007.sh <시험 폴더>
#
#   회의 한 건: 시험 폴더에 그 회의의 폰별 원본 flac 을 바로 넣는다.
#     /data/timbel/howling_test/meet_20261105/A.Biz_m_~~~.flac ...
#   회의 여러 건: 회의마다 하위 폴더.
#
#   예) ./howling_test_007.sh /data/timbel/howling_test/meet_20261105
#
#   결과는 시험 폴더의 results_007_<시각>/<회의>/ 에 쌓인다.
#
# 환경변수 (필요할 때만):
#   INSTALL_DIR   서버 설치 경로     (기본 /data/timblo/mobile-sync-server)
#   SYNC_SOUND    앱이 쓴 싱크음     (기본 beep. beep / voice / jingle 만 가능)
#   BEEP_SOURCE   누가 비프를 울렸나 (기본 all. leader / all)
#   SOUND_LANG    안내 음성 언어     (기본 ko. ko / en)
#
# 안전장치: 시작할 때 .env 를 백업하고, 끝나거나 중간에 멈춰도(Ctrl+C 포함)
# 원래 .env 로 되돌리고 서버를 다시 띄운다.
#
set -euo pipefail
shopt -s nocaseglob  # 확장자 대소문자 무관 (.flac / .FLAC)

SRC_DIR="${1:-}"
[[ -n "$SRC_DIR" && -d "$SRC_DIR" ]] || { echo "사용법: $0 <시험 폴더>"; exit 1; }
SRC_DIR="$(cd "$SRC_DIR" && pwd)"
INSTALL_DIR="${INSTALL_DIR:-/data/timblo/mobile-sync-server}"
SYNC_SOUND="${SYNC_SOUND:-beep}"
BEEP_SOURCE="${BEEP_SOURCE:-all}"
SOUND_LANG="${SOUND_LANG:-ko}"
CONTAINER="mobile-sync-server"

case "$SYNC_SOUND" in
    beep|voice|jingle) ;;
    *) echo "ERROR: SYNC_SOUND=${SYNC_SOUND} 는 틀린 값이다. beep / voice / jingle 중 하나 (비프음이면 beep)"; exit 1 ;;
esac
case "$BEEP_SOURCE" in
    leader|all) ;;
    *) echo "ERROR: BEEP_SOURCE=${BEEP_SOURCE} 는 틀린 값이다. leader / all 중 하나"; exit 1 ;;
esac

ENV_FILE="${INSTALL_DIR}/.env"
[[ -f "$ENV_FILE" ]] || { echo "ERROR: ${ENV_FILE} 가 없다. INSTALL_DIR 을 확인할 것."; exit 1; }

count_flac() { local n=0 f; for f in "$1"/*.flac; do [[ -f "$f" ]] && n=$((n + 1)); done; echo "$n"; }

MEETINGS=()
if (( $(count_flac "$SRC_DIR") >= 2 )); then
    MEETINGS=("$SRC_DIR")
else
    for d in "$SRC_DIR"/*/; do
        [[ -d "$d" ]] || continue
        d="${d%/}"
        [[ "$(basename "$d")" == results_* ]] && continue
        n=$(count_flac "$d")
        if (( n >= 2 )); then
            MEETINGS+=("$d")
        elif (( n == 1 )); then
            echo "건너뜀: $(basename "$d") — flac 이 1개뿐이다 (폰별 원본이 2개 이상 필요)"
        fi
    done
fi
(( ${#MEETINGS[@]} >= 1 )) || { echo "ERROR: 폰별 원본 flac 이 2개 이상 든 회의 폴더가 없다."; exit 1; }

if docker compose version >/dev/null 2>&1; then
    COMPOSE=(docker compose)
elif command -v docker-compose >/dev/null 2>&1; then
    COMPOSE=(docker-compose)
else
    echo "ERROR: docker compose 가 없다."; exit 1
fi

PORT="$(grep -E '^HOST_PORT=' "$ENV_FILE" | tail -1 | cut -d= -f2 | tr -d '[:space:]')"
API="http://127.0.0.1:${PORT:-9995}"
OUT_DIR="${SRC_DIR}/results_007_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$OUT_DIR"
BACKUP="${INSTALL_DIR}/.env.before_howling_test"
cp "$ENV_FILE" "$BACKUP"

log() { printf '[%s] %s\n' "$(date +%H:%M:%S)" "$*" | tee -a "$OUT_DIR/run.log"; }

restart() {
    ( cd "$INSTALL_DIR" && "${COMPOSE[@]}" --env-file .env up -d >/dev/null 2>&1 )
    for _ in $(seq 1 60); do
        curl -fs "$API/health" >/dev/null 2>&1 && return 0
        sleep 2
    done
    log "ERROR: 서버가 2분 안에 뜨지 않았다. docker logs --tail 50 ${CONTAINER} 확인"
    return 1
}

restore() {
    cp "$BACKUP" "$ENV_FILE"
    restart || true
    log "원래 .env 로 되돌리고 서버를 다시 띄웠다"
}
trap restore EXIT

# 시험 조건만 바꾸고 나머지 설정은 그대로. 업로드 한도는 시험 중에만 4GB.
grep -v -E '^(MERGE_LEVEL|ALIGN_REFINE|MERGE_UPLOAD_MAX_BYTES)=' "$BACKUP" > "$ENV_FILE"
printf 'MERGE_LEVEL=off\nALIGN_REFINE=1\nMERGE_UPLOAD_MAX_BYTES=4294967296\n' >> "$ENV_FILE"

log "회의 ${#MEETINGS[@]}건 × 1개 병합 (파형 맞추기 켬, 음량 맞추기 끔)"
for m in "${MEETINGS[@]}"; do
    log "  $(basename "$m"): $(cd "$m" && ls *.flac | tr '\n' ' ')"
done
log "syncSound=${SYNC_SOUND} beepSource=${BEEP_SOURCE} soundLang=${SOUND_LANG}"
log "결과 폴더: ${OUT_DIR}"
log "서버 다시 띄우는 중"
restart

name="merged_007"
for m in "${MEETINGS[@]}"; do
    out="$OUT_DIR/$(basename "$m")"
    mkdir -p "$out"
    args=()
    for f in "$m"/*.flac; do args+=(-F "files=@${f}"); done
    body="$(curl -s -X POST "$API/v1/merge/upload" "${args[@]}" \
        -F align=none -F "syncSound=${SYNC_SOUND}" -F "beepSource=${BEEP_SOURCE}" \
        -F "soundLang=${SOUND_LANG}" -F automix=true)"
    job="$(printf '%s' "$body" | grep -o 'mrg_[0-9a-f]*' | head -1 || true)"
    if [[ -z "$job" ]]; then
        log "  $(basename "$m"): 요청 실패 — ${body}"
        continue
    fi
    until curl -s "$API/v1/merge/$job" | grep -q -E '"status": "(DONE|ERROR)"'; do sleep 5; done
    curl -s "$API/v1/merge/$job" > "$out/${name}.json"
    if grep -q '"status": "DONE"' "$out/${name}.json"; then
        curl -s -o "$out/${name}.flac" "$API/v1/merge/$job/download"
        log "  $(basename "$m"): 완료 (${job}) → ${out}/${name}.flac"
    else
        log "  $(basename "$m"): 병합 실패 — ${name}.json 참조"
    fi
    grep -q '"refine"' "$out/${name}.json" \
        || log "  주의: 결과에 파형 맞추기 기록이 없다. 서버가 0.0.7 인지 확인할 것"
    docker logs "$CONTAINER" 2>&1 | grep "$job" | grep -E 'merge_alignment|merge_refine|merge_drift' > "$out/${name}.log" || true
done

log "끝. 결과 ${OUT_DIR} — 회의 폴더마다 ${name}.flac / .json / .log"
