#!/usr/bin/env bash
#
# howling_test.sh — 하울링·단어 반복 원인 찾기: 설정 12가지 조합을 자동으로 병합한다.
#
#   음량 맞추기(MERGE_LEVEL)   dynamic / off
#   드리프트 보정(DRIFT_CORRECT) 1 / 0
#   오토믹서                   -20dB(지금) / -60dB(세게) / 끔
#   → 2 × 2 × 3 = 12 개 결과 (회의마다)
#
# 사용법 (회사 서버에서):
#   ./howling_test.sh <시험 폴더>
#
#   회의 여러 건: 회의마다 하위 폴더를 만들고 그 회의의 폰별 원본 flac 을 넣는다.
#     /data/howling_test/회의1/phone_A.flac, phone_B.flac ...
#     /data/howling_test/회의2/...
#   회의 한 건: 시험 폴더에 바로 flac 들을 넣어도 된다.
#
#   예) ./howling_test.sh /data/howling_test
#
#   결과는 시험 폴더의 results_<시각>/<회의>/ 에 쌓인다.
#
# 환경변수 (필요할 때만):
#   INSTALL_DIR   서버 설치 경로     (기본 /data/timblo/mobile-sync-server)
#   SYNC_SOUND    앱이 쓴 싱크음     (기본 beep. beep / voice / jingle 만 가능)
#   BEEP_SOURCE   누가 비프를 울렸나 (기본 all. leader / all)
#   SOUND_LANG    안내 음성 언어     (기본 ko. ko / en)
#
# 안전장치: 시작할 때 .env 를 백업하고, 끝나거나 중간에 멈춰도(Ctrl+C 포함)
# 원래 .env 로 되돌리고 서버를 다시 띄운다.
#
set -euo pipefail
shopt -s nocaseglob  # 확장자 대소문자 무관 (.flac / .FLAC)

SRC_DIR="${1:-}"
[[ -n "$SRC_DIR" && -d "$SRC_DIR" ]] || { echo "사용법: $0 <시험 폴더>"; exit 1; }
SRC_DIR="$(cd "$SRC_DIR" && pwd)"
INSTALL_DIR="${INSTALL_DIR:-/data/timblo/mobile-sync-server}"
SYNC_SOUND="${SYNC_SOUND:-beep}"
BEEP_SOURCE="${BEEP_SOURCE:-all}"
SOUND_LANG="${SOUND_LANG:-ko}"
CONTAINER="mobile-sync-server"

case "$SYNC_SOUND" in
    beep|voice|jingle) ;;
    *) echo "ERROR: SYNC_SOUND=${SYNC_SOUND} 는 틀린 값이다. beep / voice / jingle 중 하나 (비프음이면 beep)"; exit 1 ;;
esac
case "$BEEP_SOURCE" in
    leader|all) ;;
    *) echo "ERROR: BEEP_SOURCE=${BEEP_SOURCE} 는 틀린 값이다. leader / all 중 하나"; exit 1 ;;
esac

ENV_FILE="${INSTALL_DIR}/.env"
[[ -f "$ENV_FILE" ]] || { echo "ERROR: ${ENV_FILE} 가 없다. INSTALL_DIR 을 확인할 것."; exit 1; }

count_flac() { local n=0 f; for f in "$1"/*.flac; do [[ -f "$f" ]] && n=$((n + 1)); done; echo "$n"; }

# 회의 목록: 시험 폴더에 flac 이 바로 있으면 그 폴더 하나, 아니면 flac 2개 이상인 하위 폴더 전부.
MEETINGS=()
if (( $(count_flac "$SRC_DIR") >= 2 )); then
    MEETINGS=("$SRC_DIR")
else
    for d in "$SRC_DIR"/*/; do
        [[ -d "$d" ]] || continue
        d="${d%/}"
        [[ "$(basename "$d")" == results_* ]] && continue
        n=$(count_flac "$d")
        if (( n >= 2 )); then
            MEETINGS+=("$d")
        elif (( n == 1 )); then
            echo "건너뜀: $(basename "$d") — flac 이 1개뿐이다 (폰별 원본이 2개 이상 필요)"
        fi
    done
fi
(( ${#MEETINGS[@]} >= 1 )) || { echo "ERROR: 폰별 원본 flac 이 2개 이상 든 회의 폴더가 없다."; exit 1; }

if docker compose version >/dev/null 2>&1; then
    COMPOSE=(docker compose)
elif command -v docker-compose >/dev/null 2>&1; then
    COMPOSE=(docker-compose)
else
    echo "ERROR: docker compose 가 없다."; exit 1
fi

PORT="$(grep -E '^HOST_PORT=' "$ENV_FILE" | tail -1 | cut -d= -f2 | tr -d '[:space:]')"
API="http://127.0.0.1:${PORT:-9995}"
OUT_DIR="${SRC_DIR}/results_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$OUT_DIR"
BACKUP="${INSTALL_DIR}/.env.before_howling_test"
cp "$ENV_FILE" "$BACKUP"

log() { printf '[%s] %s\n' "$(date +%H:%M:%S)" "$*" | tee -a "$OUT_DIR/run.log"; }

restart() {
    ( cd "$INSTALL_DIR" && "${COMPOSE[@]}" --env-file .env up -d >/dev/null 2>&1 )
    for _ in $(seq 1 60); do
        curl -fs "$API/health" >/dev/null 2>&1 && return 0
        sleep 2
    done
    log "ERROR: 서버가 2분 안에 뜨지 않았다. docker logs --tail 50 ${CONTAINER} 확인"
    return 1
}

restore() {
    cp "$BACKUP" "$ENV_FILE"
    restart || true
    log "원래 .env 로 되돌리고 서버를 다시 띄웠다"
}
trap restore EXIT

# 시험하는 세 가지만 지우고 새 값으로 덧붙인다. 나머지 설정은 그대로.
# 업로드 한도(기본 512MB)는 긴 회의를 여러 폰으로 녹음하면 넘으므로 시험 중에만 4GB 로 올린다.
apply_env() {  # $1 MERGE_LEVEL  $2 DRIFT_CORRECT  $3 AUTOMIX_FLOOR_DB
    grep -v -E '^(MERGE_LEVEL|DRIFT_CORRECT|AUTOMIX_FLOOR_DB|MERGE_UPLOAD_MAX_BYTES)=' "$BACKUP" > "$ENV_FILE"
    printf 'MERGE_LEVEL=%s\nDRIFT_CORRECT=%s\nAUTOMIX_FLOOR_DB=%s\nMERGE_UPLOAD_MAX_BYTES=4294967296\n' \
        "$1" "$2" "$3" >> "$ENV_FILE"
}

run_one() {  # $1 회의 폴더  $2 이름  $3 automix(true/false)
    local meeting="$1" name="$2" automix="$3" args=() job body f
    local out="$OUT_DIR/$(basename "$meeting")"
    local label="$(basename "$meeting") / ${name}"
    mkdir -p "$out"
    for f in "$meeting"/*.flac; do args+=(-F "files=@${f}"); done
    body="$(curl -s -X POST "$API/v1/merge/upload" "${args[@]}" \
        -F align=none -F "syncSound=${SYNC_SOUND}" -F "beepSource=${BEEP_SOURCE}" \
        -F "soundLang=${SOUND_LANG}" -F "automix=${automix}")"
    job="$(printf '%s' "$body" | grep -o 'mrg_[0-9a-f]*' | head -1 || true)"
    if [[ -z "$job" ]]; then
        log "  ${label}: 요청 실패 — ${body}"
        return 0
    fi
    until curl -s "$API/v1/merge/$job" | grep -q -E '"status": "(DONE|ERROR)"'; do sleep 5; done
    curl -s "$API/v1/merge/$job" > "$out/${name}.json"
    if grep -q '"status": "DONE"' "$out/${name}.json"; then
        curl -s -o "$out/${name}.flac" "$API/v1/merge/$job/download"
        log "  ${label}: 완료 (${job})"
    else
        log "  ${label}: 병합 실패 — ${name}.json 참조"
    fi
    docker logs "$CONTAINER" 2>&1 | grep "$job" | grep -E 'merge_alignment|merge_drift' > "$out/${name}.log" || true
}

write_sheet() {  # $1 회의 폴더
    local name; name="$(basename "$1")"
    cat > "$OUT_DIR/${name}/듣기표.md" <<SHEET
# 듣기 결과표 — ${name}

먼저 1번(지금 설정 그대로)과 12번(전부 끔)을 비교한다. 차이가 크면 나머지도 듣는다.

| 번호 | 파일 | 울림 (없음/약함/심함) | 단어 반복 (없음/가끔/자주) | 뒤로 갈수록 심해지나 |
|---|---|---|---|---|
| 1 | level_dynamic__drift_on__automix_20db (지금 설정) | | | |
| 2 | level_dynamic__drift_on__automix_off | | | |
| 3 | level_dynamic__drift_on__automix_60db | | | |
| 4 | level_dynamic__drift_off__automix_20db | | | |
| 5 | level_dynamic__drift_off__automix_off | | | |
| 6 | level_dynamic__drift_off__automix_60db | | | |
| 7 | level_off__drift_on__automix_20db | | | |
| 8 | level_off__drift_on__automix_off | | | |
| 9 | level_off__drift_on__automix_60db | | | |
| 10 | level_off__drift_off__automix_20db | | | |
| 11 | level_off__drift_off__automix_60db | | | |
| 12 | level_off__drift_off__automix_off (전부 끔) | | | |

회의 인원 / 폰 대수 / 폰 사이 거리 / 회의 길이:
SHEET
}

log "회의 ${#MEETINGS[@]}건 × 12가지 = $(( ${#MEETINGS[@]} * 12 ))번 병합"
for m in "${MEETINGS[@]}"; do
    log "  $(basename "$m"): $(cd "$m" && ls *.flac | tr '\n' ' ')"
done
log "syncSound=${SYNC_SOUND} beepSource=${BEEP_SOURCE} soundLang=${SOUND_LANG}"
log "결과 폴더: ${OUT_DIR}"

step=0
for level in dynamic off; do
    for drift in 1 0; do
        for floor in -20 -60; do
            step=$((step + 1))
            log "[설정 ${step}/8] MERGE_LEVEL=${level} DRIFT_CORRECT=${drift} AUTOMIX_FLOOR_DB=${floor} — 서버 다시 띄우는 중"
            apply_env "$level" "$drift" "$floor"
            restart
            d=$([[ $drift == 1 ]] && echo on || echo off)
            for m in "${MEETINGS[@]}"; do
                run_one "$m" "level_${level}__drift_${d}__automix_${floor#-}db" true
                # 오토믹서 끔은 바닥값과 무관하므로 -20 설정에서 한 번만.
                if [[ $floor == -20 ]]; then
                    run_one "$m" "level_${level}__drift_${d}__automix_off" false
                fi
            done
        done
    done
done

for m in "${MEETINGS[@]}"; do write_sheet "$m"; done

log "끝. 결과 ${OUT_DIR} — 회의 폴더마다 *.flac 12개, *.log, *.json, 듣기표.md"
log "보낼 것: 회의 폴더마다 듣기표.md + *.log + *.json, 그리고 run.log (flac 은 반출 가능할 때만)"
