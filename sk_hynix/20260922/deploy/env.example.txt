# mobile-sync-server 설정 — 이 파일을 .env 로 복사해서 쓴다.
#
#   cp .env.example .env
#
# .env 는 서버에서 손으로 관리한다. git 에 올리지 않는다.

# ---------------------------------------------------------------------
# 이미지
#   docker load 로 적재한 이미지와 정확히 일치해야 한다.
#   install.sh 를 쓰면 적재한 이미지를 읽어서 이 두 줄을 자동으로 맞춰준다.
#   직접 확인하려면:  docker images mobile-sync-server
# ---------------------------------------------------------------------
IMAGE_NAME=mobile-sync-server
IMAGE_VERSION=latest

# ---------------------------------------------------------------------
# 포트
#   HOST_PORT       서버에서 열리는 포트 (외부 접속은 이 번호)
#   CONTAINER_PORT  컨테이너 안에서 바인드하는 포트 (내부 전용, 바꿀 필요 없다)
# ---------------------------------------------------------------------
HOST_PORT=9995
CONTAINER_PORT=8080

RECOG_HOST=0.0.0.0
RECOG_PORT=8080

# ---------------------------------------------------------------------
# 컨테이너 실행 계정
#   서버 계정(aimm)의 uid:gid 를 넣는다. install.sh 가 자동으로 채운다.
#   이걸 맞춰야 호스트에서도 데이터 파일을 그대로 읽고 지울 수 있다.
#   직접 확인하려면:  id -u aimm ; id -g aimm
#
#   이미지 기본값은 10001 이지만, 그대로 쓰면 RUNTIME_DIR 안의 파일이
#   전부 10001 소유가 되어 aimm 계정으로는 손댈 수 없게 된다.
# ---------------------------------------------------------------------
RUN_UID=1000
RUN_GID=1000

# ---------------------------------------------------------------------
# 데이터 저장 위치 (호스트 경로)
#   세션 데이터 + 병합 기록(merge.db)이 여기 쌓인다.
#   이미지를 갈아끼워도 이 디렉터리는 건드리지 않는다.
#   소유권은 위 RUN_UID:RUN_GID 와 같아야 한다:
#     sudo chown -R aimm:aimm /data/timblo/mobile-sync-server-runtime
# ---------------------------------------------------------------------
RUNTIME_DIR=/data/timblo/mobile-sync-server-runtime

# ---------------------------------------------------------------------
# API 문서(/docs, /openapi.json) 접근 계정
#   브라우저로 여는 유일한 화면이라 Basic 인증을 건다.
#   /v1/merge 자체는 인증이 없다 (도커 네트워크 내부 통신).
#   SWAGGER_PASSWORD 를 빈 값으로 두면 인증이 꺼진다.
# ---------------------------------------------------------------------
SWAGGER_USER=admin
SWAGGER_PASSWORD=lena47

# ---------------------------------------------------------------------
# 음성 병합 (/v1/merge) — 전부 선택값. 비우면 아래 기본값으로 동작한다.
# ---------------------------------------------------------------------
# 동시에 실제로 병합을 돌리는 개수 (요청 접수는 이 값과 무관하게 무제한)
MERGE_WORKERS=3
# 한 건당 최대 파일 수
MERGE_MAX_FILES=5
# presigned 경로의 파일당 상한 (bytes, 기본 2GB)
MERGE_MAX_FILE_BYTES=2147483648
# 업로드 테스트 경로(/v1/merge/upload)의 합계 상한 (bytes, 기본 512MB)
MERGE_UPLOAD_MAX_BYTES=536870912
# RUNNING 진입 후 제한시간 (초). 큐 대기 시간은 포함되지 않는다.
MERGE_JOB_TIMEOUT_SEC=1800
# align=timestamp 에서 이 값을 넘는 offset 은 시계 오류로 보고 0 으로 되돌린다 (ms)
MERGE_MAX_OFFSET_MS=60000
# 작업 상태를 메모리에 캐시하는 기간 (초). 지나도 조회는 된다 (SQLite 폴백).
MERGE_JOB_TTL_SEC=86400
# SQLite 기록 보존 기간 (일). 0 이면 영구 보관.
MERGE_RECORD_TTL_DAYS=90

# ---------------------------------------------------------------------
# 동기화 비프음 (정렬 1순위) — 전부 선택값.
#
# 방장 폰이 스피커로 울린 비프음을 각 트랙에서 찾아 그 지점을 시작점으로 삼는다.
# align 값과 무관하게 항상 먼저 시도하고, 못 찾으면 align(timestamp/none)으로 내려간다.
# 음원 규격은 newDocs/assets/sync_beep.wav 와 반드시 일치해야 한다.
# ---------------------------------------------------------------------
# 0 으로 두면 비프 탐색을 건너뛰고 예전처럼 align 값만 쓴다.
BEEP_ENABLED=1
# 이중음 주파수 (Hz). 녹음이 16kHz 라 8000 미만이어야 하고, 배음 관계면 안 된다.
BEEP_FREQ_1=3000
BEEP_FREQ_2=4200
# 비프 하나의 길이 (ms)
BEEP_DURATION_MS=250
# 반복 횟수와 간격 (ms). 이 간격 패턴이 오검출을 걸러내는 장치다.
BEEP_REPEAT=3
BEEP_PERIOD_MS=1000
# 파일 앞 몇 초까지 탐색할지 (초). 늘리면 그만큼 느려진다.
BEEP_SEARCH_SEC=10
# 배경 중앙값 대비 몇 배여야 비프로 인정할지. 낮추면 검출은 늘고 오검출도 는다.
BEEP_MIN_RISE=6.0
# 합쳐진 결과에서 마지막 비프 끝까지 잘라낸다 (회의 시작 지점부터 시작하게 됨).
# 자른 양은 응답의 alignment.trimmedMs 로 알려준다. 0 으로 두면 자르지 않고
# 아래 BEEP_MUTE 로 비프 구간만 무음 처리한다.
BEEP_TRIM=1
# 자른 뒤 이만큼(초)도 안 남으면 자르기를 취소하고 무음 처리로 넘어간다.
BEEP_TRIM_MIN_REMAINDER_SEC=5
# BEEP_TRIM=0 일 때만 쓰인다. 비프 구간을 무음 처리할지.
BEEP_MUTE=1
# 무음 구간의 앞뒤 여유 (ms). 앞쪽이 큰 것은 검출 온셋이 실제보다 조금 늦기 때문.
BEEP_MUTE_LEAD_MS=150
BEEP_MUTE_TAIL_MS=100
# 비프가 이 개수 이상 트랙에서 잡혀야 비프 정렬을 쓴다 (하나뿐이면 맞출 대상이 없다)
BEEP_MIN_TRACKS=2

# ---------------------------------------------------------------------
# 로그
#
# stdout 에 JSON 한 줄씩 출력한다. Dozzle 이 JSON 을 인식해 필드별로 펼치고
# level 로 색을 입힌다.
#   {"ts":"...","level":"INFO","event":"merge_stage","message":"MIXING 60%", ...}
#
# presigned URL 은 쿼리스트링(서명)을 잘라내고 호스트+경로까지만 남긴다.
# ---------------------------------------------------------------------
# DEBUG / INFO / WARNING / ERROR. 평소에는 INFO.
# DEBUG 는 ffmpeg 전체 커맨드와 트랙별 비프 검출 상세까지 찍으므로
# 문제를 쫓을 때만 잠깐 올린다.
RECOG_LOG_LEVEL=INFO
