#!/usr/bin/env bash
#
# howling_test_101.sh — 같은 원본으로 오토믹서 설정 여러 조합을 한 번에 병합해서 비교한다.
#
# 100 과 다른 점: 운영 서버를 건드리지 않는다. .env 를 고치거나 서버를 다시 띄울 필요가 없다.
#   조합마다 시험용 컨테이너(automix-test, 포트 19995)를 잠깐 띄워 병합하고 지운다.
#   나머지 설정(음량 맞추기, 폰별 음량 등)은 운영 .env 를 그대로 따른다.
#
#   cd /data/timblo/mobile-sync-server
#   ./howling_test_101.sh /data/timbel/howling_test
#
# 기본 조합 (유지시간ms:전환기준dB) — 하나씩만 바꾼 것도 넣어 무엇이 효과 있는지 갈라 본다:
#   300:3    0.0.8 값 (참고용 — 전환이 아주 많았던 값)
#   1000:6   지금 운영값 (1.0.0~) — 기준
#   2000:6   유지시간만 올림
#   1000:9   전환기준만 올림
#   2000:9   둘 다 올림 (추천 후보)
#   3000:12  아주 강하게 (전환은 거의 없지만, 말하는 사람이 바뀌어도 늦게 따라감)
#   조합 바꾸기:  COMBOS="2000:9 2500:9 2000:10" ./howling_test_101.sh <시험 폴더>
#
# 하는 일 (회의마다):
#   ① 원본 분석 — 처음 한 번만 (이미 있으면 건너뜀). 100 과 같다.
#   ② 조합마다 병합 1건. 결과 이름 = 시험 컨테이너가 실제로 읽은 설정값
#        예) h2000_s9_dynamic_gain0.flac  (이미 있으면 건너뜀, 다시 하려면 FORCE=1)
#   끝에 쌓인 결과 전체 요약 (조합별 전환 횟수 / 폰별 들린 비율) → 사진으로 찍어 오면 된다.
#
# 환경변수 (필요할 때만):
#   COMBOS        시험할 조합         (기본 위 6개)
#   INSTALL_DIR   서버 설치 경로     (기본 /data/timblo/mobile-sync-server)
#   SYNC_SOUND    앱이 쓴 싱크음     (기본 beep. beep / voice / jingle 만 가능)
#   BEEP_SOURCE   누가 비프를 울렸나 (기본 all. leader / all)
#   SOUND_LANG    안내 음성 언어     (기본 ko. ko / en)
#   TEST_PORT     시험 컨테이너 포트 (기본 19995, 운영 9995 와 겹치지 않게)
#   FORCE=1       이미 있는 결과도 다시 병합
#
# 업로드 한도는 시험 컨테이너에만 4GB 로 열어 둔다 (운영 설정은 그대로).
# 병합은 한 조합씩 차례로 돈다 — 1시간 회의 6조합이면 시간이 꽤 걸린다.
#
set -euo pipefail
shopt -s nocaseglob  # 확장자 대소문자 무관 (.flac / .FLAC)

SRC_DIR="${1:-}"
[[ -n "$SRC_DIR" && -d "$SRC_DIR" ]] || { echo "사용법: $0 <시험 폴더>"; exit 1; }
SRC_DIR="$(cd "$SRC_DIR" && pwd)"
INSTALL_DIR="${INSTALL_DIR:-/data/timblo/mobile-sync-server}"
SYNC_SOUND="${SYNC_SOUND:-beep}"
BEEP_SOURCE="${BEEP_SOURCE:-all}"
SOUND_LANG="${SOUND_LANG:-ko}"
TEST_NAME="automix-test"
TEST_PORT="${TEST_PORT:-19995}"
COMBOS="${COMBOS:-300:3 1000:6 2000:6 1000:9 2000:9 3000:12}"

case "$SYNC_SOUND" in
    beep|voice|jingle) ;;
    *) echo "ERROR: SYNC_SOUND=${SYNC_SOUND} 는 틀린 값이다. beep / voice / jingle 중 하나 (비프음이면 beep)"; exit 1 ;;
esac
case "$BEEP_SOURCE" in
    leader|all) ;;
    *) echo "ERROR: BEEP_SOURCE=${BEEP_SOURCE} 는 틀린 값이다. leader / all 중 하나"; exit 1 ;;
esac

ENV_FILE="${INSTALL_DIR}/.env"
[[ -f "$ENV_FILE" ]] || { echo "ERROR: ${ENV_FILE} 가 없다. INSTALL_DIR 을 확인할 것."; exit 1; }

count_flac() { local n=0 f; for f in "$1"/*.flac; do [[ -f "$f" ]] && n=$((n + 1)); done; echo "$n"; }

MEETINGS=()
if (( $(count_flac "$SRC_DIR") >= 2 )); then
    MEETINGS=("$SRC_DIR")
else
    for d in "$SRC_DIR"/*/; do
        [[ -d "$d" ]] || continue
        d="${d%/}"
        [[ "$(basename "$d")" == results_* ]] && continue
        n=$(count_flac "$d")
        if (( n >= 2 )); then
            MEETINGS+=("$d")
        elif (( n == 1 )); then
            echo "건너뜀: $(basename "$d") — flac 이 1개뿐이다 (폰별 원본이 2개 이상 필요)"
        fi
    done
fi
(( ${#MEETINGS[@]} >= 1 )) || { echo "ERROR: 폰별 원본 flac 이 2개 이상 든 회의 폴더가 없다."; exit 1; }

# 줄이 없어도 멈추지 않게 (|| true). set -e 에서 grep 실패는 조용히 스크립트를 끝낸다.
env_get() { { grep -E "^$1=" "$ENV_FILE" || true; } | tail -1 | cut -d= -f2- | tr -d '[:space:]'; }
API="http://127.0.0.1:${TEST_PORT}"
IMAGE="${IMAGE:-$(env_get IMAGE_NAME):$(env_get IMAGE_VERSION)}"
[[ "$IMAGE" == ":"* ]] && IMAGE="mobile-sync-server${IMAGE}"
IMAGE="${IMAGE%:}"
docker image inspect "$IMAGE" >/dev/null 2>&1 \
    || { echo "ERROR: 이미지 ${IMAGE} 가 없다. (docker images mobile-sync-server 로 확인)"; exit 1; }
OUT_DIR="${SRC_DIR}/results_101"
mkdir -p "$OUT_DIR"

log() { printf '[%s] %s\n' "$(date +%H:%M:%S)" "$*" | tee -a "$OUT_DIR/run.log"; }

stop_test() { docker rm -f "$TEST_NAME" >/dev/null 2>&1 || true; }
trap stop_test EXIT

# 시험 컨테이너 띄우기: 운영 .env 를 깔고 두 값만 덮어쓴다. 실제 읽은 설정으로 NAME 을 만든다.
start_test() {  # $1 유지ms  $2 기준dB
    stop_test
    docker run -d --name "$TEST_NAME" --env-file "$ENV_FILE" \
        -e RECOG_HOST=0.0.0.0 -e RECOG_PORT=8080 \
        -e MERGE_AUTOMIX=true -e AUTOMIX_HOLD_MS="$1" -e AUTOMIX_SWITCH_DB="$2" \
        -e MERGE_UPLOAD_MAX_BYTES=4294967296 -e MERGE_WORKERS=1 \
        -p "127.0.0.1:${TEST_PORT}:8080" \
        "$IMAGE" --data-root /var/lib/recog/runtime serve >/dev/null
    local i
    for i in $(seq 1 30); do
        curl -fs --max-time 3 "$API/health" >/dev/null 2>&1 && break
        sleep 1
    done
    curl -fs --max-time 3 "$API/health" >/dev/null 2>&1 \
        || { docker logs --tail 30 "$TEST_NAME"; echo "ERROR: 시험 컨테이너가 뜨지 않았다."; exit 1; }
    SETTINGS="$(docker exec "$TEST_NAME" python -c '
from recog import automix, merge_job
print(automix.MODE, automix.HOLD_MS, int(automix.SWITCH_DB), merge_job.LEVEL_MODE,
      int(getattr(automix, "TRACK_GAIN_DEFAULT", False)))' 2>/dev/null || true)"
    read -r MODE HOLD SWITCH LEVEL GAIN <<< "${SETTINGS:-? ? ? ? ?}"
    NAME="h${HOLD}_s${SWITCH}_${LEVEL}_gain${GAIN}"
    [[ "$MODE" == "select" ]] || NAME="${MODE}_${NAME}"
}

for m in "${MEETINGS[@]}"; do
    log "회의 $(basename "$m"): $(cd "$m" && ls *.flac | tr '\n' ' ')"
done
log "이미지: ${IMAGE}   조합: ${COMBOS}"
log "syncSound=${SYNC_SOUND} beepSource=${BEEP_SOURCE} soundLang=${SOUND_LANG}  결과 폴더: ${OUT_DIR}"

# ---- ① 원본 분석 (처음 한 번만) --------------------------------------------------
# 서버 이미지 안의 ffmpeg / python 을 쓴다 (서버 호스트에 없어도 된다). 네트워크 없이, 읽기 전용으로.
ANALYZE_PY="$OUT_DIR/.analyze.py"
cat > "$ANALYZE_PY" <<'PY'
import array, math, sys
# 표준입력: 16kHz 모노 s16le. 50ms 마다 크기(dB)를 재서 1분 단위로 묶는다.
# 말소리 = 그 1분에서 큰 쪽 10% 지점, 조용할 때 = 작은 쪽 10% 지점, 차이 = 둘의 차.
# 차이가 작을수록 말 사이에도 소리(울림·잡음)가 차 있다는 뜻.
name, rate = sys.argv[1], int(sys.argv[2])
FRAME, PER_MIN = 800, 1200
levels, buf = [], b""
while True:
    chunk = sys.stdin.buffer.read(FRAME * 2 * 256)
    if not chunk:
        break
    buf += chunk
    usable = len(buf) // (FRAME * 2) * (FRAME * 2)
    samples = array.array("h", buf[:usable])
    buf = buf[usable:]
    for i in range(0, len(samples), FRAME):
        part = samples[i:i + FRAME]
        power = sum(s * s for s in part) / len(part)
        levels.append(10 * math.log10(power / 32768.0 ** 2) if power > 0 else -100.0)
def pct(values, p):
    ordered = sorted(values)
    return ordered[min(len(ordered) - 1, int(len(ordered) * p))]
def stats(values):
    if len(values) < 20:
        return None
    loud, quiet = pct(values, 0.9), pct(values, 0.1)
    return loud, quiet, loud - quiet
minutes = [levels[i:i + PER_MIN] for i in range(0, len(levels), PER_MIN)]
print(f"## {name}  샘플레이트 {rate} Hz  길이 {len(levels) * 0.05 / 60:.1f}분")
print("분 | 말소리 dB | 조용할때 dB | 차이 dB")
for index, values in enumerate(minutes):
    result = stats(values)
    if result:
        print(f"{index + 1:3d} | {result[0]:6.1f} | {result[1]:6.1f} | {result[2]:5.1f}")
def span(values):
    result = stats(values)
    return "  -  " if result is None else f"말 {result[0]:6.1f} 조용 {result[1]:6.1f} 차이 {result[2]:5.1f}"
n = len(minutes)
head = [v for m in minutes[:5] for v in m]
tail = [v for m in minutes[-5:] for v in m] if n > 10 else []
middle = [v for m in minutes[5:-5] for v in m] if n > 10 else []
print(f"SUMMARY {name} | {rate}Hz | {n}분 | 처음5분 {span(head)} | 중간 {span(middle)} | 마지막5분 {span(tail)}")
print()
PY
chmod 0644 "$ANALYZE_PY"

analyze_meeting() {  # $1 회의 폴더  $2 결과 폴더
    local m="$1" out="$2" f base rate i=1
    : > "$out/원본분석.txt"
    for f in "$m"/*.flac; do
        base="$(basename "$f")"
        rate="$(docker run --rm --network none --user 0 --entrypoint ffprobe -v "$m:/in:ro" "$IMAGE" \
            -v error -select_streams a:0 -show_entries stream=sample_rate -of csv=p=0 "/in/$base" 2>/dev/null | head -1 || true)"
        log "    f${i} = ${base} (${rate:-?} Hz) 분석 중"
        echo "f${i} = ${base}" >> "$out/원본분석.txt"
        docker run --rm --network none --user 0 --entrypoint sh \
            -v "$m:/in:ro" -v "$ANALYZE_PY:/a.py:ro" "$IMAGE" -c \
            "ffmpeg -v error -i \"/in/$base\" -ac 1 -ar 16000 -f s16le - | python3 /a.py f${i} ${rate:-0}" \
            >> "$out/원본분석.txt" 2>> "$OUT_DIR/run.log" \
            || log "    f${i}: 분석 실패 (run.log 참조)"
        i=$((i + 1))
    done
}

for m in "${MEETINGS[@]}"; do
    out="$OUT_DIR/$(basename "$m")"
    mkdir -p "$out"
    if grep -q '^SUMMARY' "$out/원본분석.txt" 2>/dev/null; then
        log "[①] $(basename "$m"): 원본 분석 이미 있음 — 건너뜀"
    elif ! docker image inspect "$IMAGE" >/dev/null 2>&1; then
        log "[①] $(basename "$m"): 원본 분석 건너뜀 — 이미지 ${IMAGE} 가 없다"
    else
        log "[①] $(basename "$m"): 원본 분석"
        analyze_meeting "$m" "$out"
    fi
done

# ---- ② 병합 1건 (지금 서버 설정 그대로) --------------------------------------------
run_one() {  # $1 결과 폴더  $2 결과 이름  나머지 = 올릴 flac 파일들
    local out="$1" name="$2" body job f
    shift 2
    local args=()
    for f in "$@"; do args+=(-F "files=@${f}"); done
    log "    ${name}: 올리는 중 (파일 $#개)"
    body="$(curl -s -X POST "$API/v1/merge/upload" "${args[@]}" \
        -F align=none -F "syncSound=${SYNC_SOUND}" -F "beepSource=${BEEP_SOURCE}" \
        -F "soundLang=${SOUND_LANG}" -F automix=true)"
    job="$(printf '%s' "$body" | grep -o 'mrg_[0-9a-f]*' | head -1 || true)"
    if [[ -z "$job" ]]; then
        log "    ${name}: 요청 실패 — ${body}"
        return 0
    fi
    local waited=0
    log "    ${name}: 병합 중 (${job})"
    until curl -s --max-time 10 "$API/v1/merge/$job" | grep -q -E '"status": "(DONE|ERROR)"'; do
        sleep 5
        waited=$((waited + 5))
        if (( waited % 60 == 0 )); then log "    ${name}: ... 병합 기다리는 중 $((waited / 60))분"; fi
    done
    curl -s "$API/v1/merge/$job" > "$out/${name}.json"
    if grep -q '"status": "DONE"' "$out/${name}.json"; then
        curl -s -o "$out/${name}.flac" "$API/v1/merge/$job/download"
        log "    ${name}: 완료 (${job})"
    else
        log "    ${name}: 병합 실패 — ${name}.json 참조"
    fi
    docker logs "$TEST_NAME" 2>&1 | grep "$job" \
        | grep -E 'merge_alignment|merge_drift|merge_track_gain|merge_automix' > "$out/${name}.log" || true
}

for combo in $COMBOS; do
    hold="${combo%%:*}"; db="${combo##*:}"
    log "[②] 조합 유지 ${hold}ms / 기준 ${db}dB — 시험 컨테이너 띄우는 중"
    start_test "$hold" "$db"
    log "    실제 읽은 설정: MODE=${MODE} HOLD=${HOLD} SWITCH=${SWITCH} MERGE_LEVEL=${LEVEL} GAIN=${GAIN} → ${NAME}"
    for m in "${MEETINGS[@]}"; do
        out="$OUT_DIR/$(basename "$m")"
        if [[ -f "$out/${NAME}.flac" && "${FORCE:-0}" != "1" ]]; then
            log "    $(basename "$m"): ${NAME} 이미 있음 — 건너뜀 (다시 하려면 FORCE=1)"
            continue
        fi
        log "    $(basename "$m"): ${NAME}"
        run_one "$out" "$NAME" "$m"/*.flac
    done
    stop_test
done

# ---- 요약 (지금까지 쌓인 결과 전부) -------------------------------------------------
{
    echo
    echo "================ 요약 (이 화면을 사진으로 찍어 오면 된다) ================"
    for m in "${MEETINGS[@]}"; do
        out="$OUT_DIR/$(basename "$m")"
        echo "[$(basename "$m")]"
        grep -h '^f[0-9]* = ' "$out/원본분석.txt" 2>/dev/null || true
        grep -h '^SUMMARY' "$out/원본분석.txt" 2>/dev/null | sed 's/^SUMMARY //' || true
        for log_file in "$out"/*.log; do
            [[ -f "$log_file" ]] || continue
            n="$(basename "$log_file" .log)"
            echo "${n}: $(grep -o '"heardPct": {[^}]*}, "switches": [0-9]*' "$log_file" | head -1)"
            gain="$(grep -o '"gainDb": {[^}]*}' "$log_file" | head -1 || true)"
            [[ -n "$gain" ]] && echo "    ${gain}"
            mode="$(grep merge_alignment "$log_file" | grep -o '"mode": "[a-z]*"' | head -1 || true)"
            [[ "$mode" == '"mode": "beep"' ]] || echo "    주의: 정렬 ${mode:-기록 없음} (beep 이 아니다)"
        done
    done
    echo "==========================================================================="
} | tee -a "$OUT_DIR/run.log"

log "끝. 결과 ${OUT_DIR}/<회의>/h*_s*.flac — 같은 구간을 들어보며 비교. 조합을 더 하려면 COMBOS 로 다시 돌리면 옆에 쌓인다"
